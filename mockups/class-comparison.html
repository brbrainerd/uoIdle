    if (sm.PerfectionTarget != defender) { sm.PerfectionTarget = defender; sm.PerfectionBonus = 0; }
    sm.PerfectionBonus = Math.Min(15, sm.PerfectionBonus + (int)(sm.Skills.Bushido.Value / 25));
